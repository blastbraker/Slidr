# How to set your Unsplash API Key (for Windows PowerShell):
# 
# $env:UNSPLASH_ACCESS_KEY = "your_key_here"
#
# Then run Slidr:
# python -m slidr.gui
#
# Or on Windows Command Prompt:
# set UNSPLASH_ACCESS_KEY=your_key_here
# python -m slidr.gui
#
# Or on Mac/Linux terminal:
# export UNSPLASH_ACCESS_KEY=your_key_here
# python -m slidr.gui

# Your key will NOT be saved to any file - it stays in your environment!
